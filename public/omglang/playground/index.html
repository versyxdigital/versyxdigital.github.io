Basics

          

            Variables
            
alloc x := 10       # introduce
x := x + 1          # change
emit x              # print

          

          

            Numbers
            
emit 7 / 2           # 3.5   true division
emit 7 // 2          # 3     floor division
emit 7 % 2           # 1     remainder
emit 6 & 3            # 2     bitwise: also | ^ ~ << >>

          

          

            Strings
            
alloc s := "hello"
emit s + ", world"   # join
emit s[0]            # h
emit s[1:3]          # el
emit length(s)       # 5

          

          

            Conditions
            
if n < 0 {
    emit "negative"
} elif n == 0 {
    emit "zero"
} else {
    emit "positive"
}
# and, or, not, ==, !=, <, <=, >, >=

          

          

            Loops
            
alloc i := 0
loop i < 5 {
    if i == 3 { break }
    emit i
    i := i + 1
}
for x in [1, 2, 3] {
    if x == 2 { continue }
    emit x
}

          


          
Data

          

            Lists
            
alloc xs := [1, 2, 3]
append(xs, 4)        # add to the end
xs[0] := 99          # replace
emit xs[-1]          # last item
emit xs[1:3]         # slice
emit length(xs)

          

          

            Dictionaries
            
alloc d := {name: "Chris", age: 32}
emit d.name
d["job"] := "engineer"
emit dict_keys(d)
emit has_key(d, "age")

          


          
Functions

          

            Define and call
            
proc add(a, b) {
    return a + b
}
emit add(2, 3)

          

          

            Closures
            
proc make_counter() {
    alloc n := 0
    proc tick() {
        n := n + 1
        return n
    }
    return tick
}

          


          
Errors

          

            Catch, raise, assert
            
try {
    panic("boom")
} except err {
    emit "caught: " + err
}
facts 1 + 1 == 2     # stops the program if false
